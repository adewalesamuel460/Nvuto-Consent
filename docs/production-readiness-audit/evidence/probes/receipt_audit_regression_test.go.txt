package service

import (
    "testing"
    "time"

    "github.com/Atlasfsp/Nvuto-Consent/internal/domain"
    "github.com/jackc/pgx/v5/pgtype"
)

// This is an audit-only probe of the real pgx timestamptz binary codec.
// It does not connect to a database and does not claim a YSQL end-to-end result.
func TestAuditReceiptHashSurvivesPGXTimestampRoundTrip(t *testing.T) {
    issued := time.Date(2026, 10, 19, 12, 0, 0, 123456789, time.UTC)
    consent := domain.Consent{ID: "audit-consent", TenantID: "audit-tenant", SubjectID: "audit-subject", Purpose: "marketing", Channel: "email", NoticeVersion: "audit-notice", Jurisdiction: "NG"}
    receipt := makeReceipt(consent, domain.Evidence{Method: "audit", ArtifactHash: "synthetic-evidence", CollectedAt: issued.Add(-time.Hour)}, issued)
    codec := pgtype.NewMap()
    encoded, err := codec.Encode(pgtype.TimestamptzOID, pgtype.BinaryFormatCode, receipt.IssuedAt, nil)
    if err != nil { t.Fatal(err) }
    var decoded time.Time
    if err := codec.Scan(pgtype.TimestamptzOID, pgtype.BinaryFormatCode, encoded, &decoded); err != nil { t.Fatal(err) }
    receipt.IssuedAt = decoded.UTC()
    if receiptHash(receipt) != receipt.ReceiptHash {
        t.Fatalf("receipt hash no longer verifies after actual pgx timestamptz encode/decode: original=%s decoded=%s; live database was not used", issued.Format(time.RFC3339Nano), receipt.IssuedAt.Format(time.RFC3339Nano))
    }
}
